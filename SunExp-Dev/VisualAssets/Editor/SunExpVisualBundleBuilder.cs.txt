// Copy this file into a Unity Editor project as Assets/Editor/SunExpVisualBundleBuilder.cs.
// Keep the asset paths aligned with SunExp-Dev/VisualAssets/sunexp_visuals.pipeline.json
// and SunExp/visual.registry.json.

using System;
using System.IO;
using UnityEditor;
using UnityEngine;

public static class SunExpVisualBundleBuilder
{
    private const string BundleName = "sunexp_visuals";
    private const string StarScoreShaderAssetPath = "Assets/SunExp/Visuals/Shaders/StarScoreHud.shader";
    private const string MaterialAssetPath = "Assets/SunExp/Visuals/Materials/StarScoreHudLit.mat";
    private const string StarScoreMaterialAssetPath = MaterialAssetPath;
    private const string WunaOrbitFireShaderAssetPath = "Assets/SunExp/Visuals/Shaders/WunaOrbitFire.shader";
    private const string WunaOrbitFireBackMaterialAssetPath = "Assets/SunExp/Visuals/Materials/WunaOrbitFireBack.mat";
    private const string WunaOrbitFireFrontMaterialAssetPath = "Assets/SunExp/Visuals/Materials/WunaOrbitFireFront.mat";
    private const string BuildOutputDirectory = "Build/SunExpVisualBundles";
    private const string ModOutputPath = "../SunExp/ModResource/VisualBundles/sunexp_visuals";

    [MenuItem("SunExp/Visuals/Build Visual Bundle")]
    public static void BuildVisualBundle()
    {
        EnsureStarScoreMaterial();
        EnsureWunaOrbitFireMaterials();

        var starScoreMaterial = AssetImporter.GetAtPath(StarScoreMaterialAssetPath);
        var wunaBackMaterial = AssetImporter.GetAtPath(WunaOrbitFireBackMaterialAssetPath);
        var wunaFrontMaterial = AssetImporter.GetAtPath(WunaOrbitFireFrontMaterialAssetPath);
        if (starScoreMaterial == null || wunaBackMaterial == null || wunaFrontMaterial == null)
        {
            throw new FileNotFoundException("One or more SunExp material assets are missing.");
        }

        starScoreMaterial.assetBundleName = BundleName;
        wunaBackMaterial.assetBundleName = BundleName;
        wunaFrontMaterial.assetBundleName = BundleName;

        Directory.CreateDirectory(BuildOutputDirectory);
        BuildPipeline.BuildAssetBundles(
            BuildOutputDirectory,
            BuildAssetBundleOptions.ChunkBasedCompression,
            EditorUserBuildSettings.activeBuildTarget);

        var builtBundle = Path.Combine(BuildOutputDirectory, BundleName);
        if (!File.Exists(builtBundle))
        {
            throw new FileNotFoundException("SunExp visual bundle was not produced.", builtBundle);
        }

        var outputPath = ResolveModOutputPath();
        Directory.CreateDirectory(Path.GetDirectoryName(outputPath));
        File.Copy(builtBundle, outputPath, true);
        AssetDatabase.Refresh();
        Debug.Log("Built SunExp visual bundle: " + outputPath);
    }

    private static string ResolveModOutputPath()
    {
        var explicitPath = Environment.GetEnvironmentVariable("SUNEXP_VISUAL_BUNDLE_OUTPUT");
        return string.IsNullOrWhiteSpace(explicitPath) ? ModOutputPath : explicitPath;
    }

    private static void EnsureStarScoreMaterial()
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(StarScoreShaderAssetPath);
        if (shader == null)
        {
            throw new FileNotFoundException("SunExp shader asset is missing.", StarScoreShaderAssetPath);
        }

        var material = AssetDatabase.LoadAssetAtPath<Material>(StarScoreMaterialAssetPath);
        if (material == null)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(StarScoreMaterialAssetPath));
            material = new Material(shader)
            {
                name = "StarScoreHudLit"
            };
            AssetDatabase.CreateAsset(material, StarScoreMaterialAssetPath);
        }
        else
        {
            material.shader = shader;
        }

        material.SetFloat("_SunExpFlowSpeed", 0.55f);
        material.SetFloat("_SunExpFlowScale", 1.2f);
        material.SetFloat("_SunExpEdgeGlow", 0.35f);
        material.SetColor("_SunExpGlowColor", new Color(1f, 0.878f, 0.541f, 1f));
        material.SetColor("_SunExpFlowColor", new Color(0.616f, 0.863f, 1f, 1f));
        EditorUtility.SetDirty(material);
        AssetDatabase.SaveAssets();
    }

    private static void EnsureWunaOrbitFireMaterials()
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(WunaOrbitFireShaderAssetPath);
        if (shader == null)
        {
            throw new FileNotFoundException("SunExp WuNa orbit fire shader asset is missing.", WunaOrbitFireShaderAssetPath);
        }

        var back = EnsureMaterial(WunaOrbitFireBackMaterialAssetPath, shader, "WunaOrbitFireBack");
        back.SetFloat("_SunExpLayer", -1f);
        back.SetFloat("_SunExpNoiseScale", 3.15f);
        back.SetFloat("_SunExpDistortion", 0.11f);
        back.SetFloat("_SunExpAlphaCutoff", 0.16f);
        back.SetFloat("_SunExpAlphaSoftness", 0.11f);
        back.SetFloat("_SunExpFlowSpeed", 0.28f);
        back.SetColor("_SunExpCoreColor", new Color(1f, 0.82f, 0.34f, 0.72f));
        back.SetColor("_SunExpEdgeColor", new Color(0.96f, 0.18f, 0.045f, 0.62f));
        back.SetColor("_SunExpSmokeColor", new Color(0.2f, 0.02f, 0.012f, 0.18f));

        var front = EnsureMaterial(WunaOrbitFireFrontMaterialAssetPath, shader, "WunaOrbitFireFront");
        front.SetFloat("_SunExpLayer", 1f);
        front.SetFloat("_SunExpNoiseScale", 3.8f);
        front.SetFloat("_SunExpDistortion", 0.16f);
        front.SetFloat("_SunExpAlphaCutoff", 0.15f);
        front.SetFloat("_SunExpAlphaSoftness", 0.095f);
        front.SetFloat("_SunExpFlowSpeed", 0.38f);
        front.SetColor("_SunExpCoreColor", new Color(1f, 0.96f, 0.58f, 0.96f));
        front.SetColor("_SunExpEdgeColor", new Color(1f, 0.22f, 0.035f, 0.84f));
        front.SetColor("_SunExpSmokeColor", new Color(0.24f, 0.026f, 0.014f, 0.2f));

        EditorUtility.SetDirty(back);
        EditorUtility.SetDirty(front);
        AssetDatabase.SaveAssets();
    }

    private static Material EnsureMaterial(string path, Shader shader, string materialName)
    {
        var material = AssetDatabase.LoadAssetAtPath<Material>(path);
        if (material == null)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            material = new Material(shader)
            {
                name = materialName
            };
            AssetDatabase.CreateAsset(material, path);
        }
        else
        {
            material.shader = shader;
        }

        return material;
    }
}
