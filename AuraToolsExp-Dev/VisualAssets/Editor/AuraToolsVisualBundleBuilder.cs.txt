using System;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEditor.Rendering;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;
using UnityEngine.UI;

public static class AuraToolsVisualBundleBuilder
{
    private const string BundleName = "auratools_visuals";
    private const string ImageShaderPath = "Assets/AuraToolsExp/Visuals/Shaders/CardFaceEffect.shader";
    private const string ImageMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/CardFrameEffectUI.mat";
    private const string MeshShaderPath = "Assets/AuraToolsExp/Visuals/Shaders/CardFrameEffectUrp.shader";
    private const string MeshMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/CardFrameEffectURP.mat";
    private const string LegacyMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/CardFaceEffect.mat";
    private const string UrpAssetPath = "Assets/AuraToolsExp/Visuals/Pipeline/AuraToolsURP.asset";
    private const string UrpRendererPath = "Assets/AuraToolsExp/Visuals/Pipeline/AuraToolsURPRenderer.asset";
    private const string LumaKeyShaderPath = "Assets/AuraToolsExp/Visuals/Shaders/AuraCgLumaKeyUI.shader";
    private const string LumaKeyMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/AuraCgLumaKeyUI.mat";
    private const string MaskedInvertShaderPath = "Assets/AuraToolsExp/Visuals/Shaders/AuraCgMaskedInvertFlash.shader";
    private const string MaskedInvertMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/AuraCgMaskedInvertFlash.mat";
    private const string ScreenFlashShaderPath = "Assets/AuraToolsExp/Visuals/Shaders/AuraCgScreenBwFlash.shader";
    private const string ScreenFlashMaterialPath = "Assets/AuraToolsExp/Visuals/Materials/AuraCgScreenBwFlash.mat";

    public static void BuildVisualBundle()
    {
        if (EditorUserBuildSettings.activeBuildTarget != BuildTarget.StandaloneWindows64
            && !EditorUserBuildSettings.SwitchActiveBuildTarget(
                BuildTargetGroup.Standalone,
                BuildTarget.StandaloneWindows64))
            throw new InvalidOperationException("AuraTools visual bundle requires StandaloneWindows64.");
        EnsureUrpPipeline();
        var cardImageMaterial = EnsureMaterial(ImageShaderPath, ImageMaterialPath, "CardFrameEffectUI");
        var cardMeshMaterial = EnsureMaterial(MeshShaderPath, MeshMaterialPath, "CardFrameEffectURP");
        var lumaKey = EnsureMaterial(LumaKeyShaderPath, LumaKeyMaterialPath, "AuraCgLumaKeyUI");
        lumaKey.SetFloat("_AuraCgKeyThreshold", 0.02f);
        lumaKey.SetFloat("_AuraCgKeySoftness", 0.1f);
        var maskedInvert = EnsureMaterial(MaskedInvertShaderPath, MaskedInvertMaterialPath, "AuraCgMaskedInvertFlash");
        maskedInvert.SetFloat("_AuraCgFlashStrength", 0.88f);
        maskedInvert.SetFloat("_AuraCgKeyThreshold", 0.02f);
        maskedInvert.SetFloat("_AuraCgKeySoftness", 0.1f);
        var screenFlash = EnsureMaterial(ScreenFlashShaderPath, ScreenFlashMaterialPath, "AuraCgScreenBwFlash");
        screenFlash.SetFloat("_AuraCgFlashStrength", 0.9f);
        EditorUtility.SetDirty(cardImageMaterial);
        EditorUtility.SetDirty(cardMeshMaterial);
        EditorUtility.SetDirty(lumaKey);
        EditorUtility.SetDirty(maskedInvert);
        EditorUtility.SetDirty(screenFlash);
        AssetDatabase.SaveAssets();
        if (AssetDatabase.LoadAssetAtPath<Material>(LegacyMaterialPath) != null)
            AssetDatabase.DeleteAsset(LegacyMaterialPath);
        AssignBundleName(ImageMaterialPath);
        AssignBundleName(MeshMaterialPath);
        AssignBundleName(LumaKeyMaterialPath);
        AssignBundleName(MaskedInvertMaterialPath);
        AssignBundleName(ScreenFlashMaterialPath);

        const string buildDirectory = "Build/AuraToolsVisualBundles";
        Directory.CreateDirectory(buildDirectory);
        BuildPipeline.BuildAssetBundles(
            buildDirectory,
            new[]
            {
                new AssetBundleBuild
                {
                    assetBundleName = BundleName,
                    assetNames = new[]
                    {
                        ImageMaterialPath,
                        MeshMaterialPath,
                        LumaKeyMaterialPath,
                        MaskedInvertMaterialPath,
                        ScreenFlashMaterialPath
                    }
                }
            },
            BuildAssetBundleOptions.ChunkBasedCompression | BuildAssetBundleOptions.ForceRebuildAssetBundle,
            EditorUserBuildSettings.activeBuildTarget);
        var built = Path.Combine(buildDirectory, BundleName);
        if (!File.Exists(built)) throw new FileNotFoundException("AuraTools visual bundle was not produced.", built);
        var output = Environment.GetEnvironmentVariable("AURATOOLS_VISUAL_BUNDLE_OUTPUT");
        if (string.IsNullOrWhiteSpace(output)) output = "../AuraToolsExp/ModResource/VisualBundles/auratools_visuals";
        ValidateBuiltBundle(built);
        Directory.CreateDirectory(Path.GetDirectoryName(output));
        File.Copy(built, output, true);
        Debug.Log("Built AuraTools visual bundle: " + output);
    }

    private static void EnsureUrpPipeline()
    {
        Directory.CreateDirectory(Path.GetDirectoryName(UrpAssetPath));
        var renderer = AssetDatabase.LoadAssetAtPath<UniversalRendererData>(UrpRendererPath);
        if (renderer == null)
        {
            renderer = ScriptableObject.CreateInstance<UniversalRendererData>();
            AssetDatabase.CreateAsset(renderer, UrpRendererPath);
        }
        var pipeline = AssetDatabase.LoadAssetAtPath<UniversalRenderPipelineAsset>(UrpAssetPath);
        if (pipeline == null)
        {
            pipeline = UniversalRenderPipelineAsset.Create(renderer);
            AssetDatabase.CreateAsset(pipeline, UrpAssetPath);
        }
        GraphicsSettings.defaultRenderPipeline = pipeline;
        QualitySettings.renderPipeline = pipeline;
        EditorUtility.SetDirty(renderer);
        EditorUtility.SetDirty(pipeline);
        AssetDatabase.SaveAssets();
    }

    private static void ValidateBuiltBundle(string output)
    {
        var bundle = AssetBundle.LoadFromFile(output);
        if (bundle == null) throw new InvalidDataException("AuraTools visual bundle cannot be loaded after build.");
        try
        {
            var imageMaterial = ValidateBuiltMaterial(
                bundle,
                ImageMaterialPath,
                "AuraTools/CardFrameEffectUI",
                "Default");
            var meshMaterial = ValidateBuiltMaterial(
                bundle,
                MeshMaterialPath,
                "AuraTools/CardFrameEffectURP",
                "CardFrameEffectURP");
            ValidateCardFrameEffectUiPixels(imageMaterial);
            ValidateCardFrameEffectUrpPixels(meshMaterial);
        }
        finally
        {
            bundle.Unload(true);
        }
    }

    private static Material ValidateBuiltMaterial(
        AssetBundle bundle,
        string materialPath,
        string shaderName,
        string requiredPass)
    {
        var material = bundle.LoadAsset<Material>(materialPath);
        if (material == null || material.shader == null || !string.Equals(material.shader.name, shaderName, StringComparison.Ordinal))
            throw new InvalidDataException("AuraTools visual bundle material is invalid: " + materialPath);
        if (!material.shader.isSupported || material.passCount <= 0)
            throw new InvalidDataException("AuraTools visual bundle shader is unsupported: " + shaderName);
        if (!string.Equals(material.GetTag("RenderPipeline", false, ""), "UniversalPipeline", StringComparison.Ordinal))
            throw new InvalidDataException("AuraTools visual bundle material is not URP-compatible: " + materialPath);
        if (material.FindPass(requiredPass) < 0)
            throw new InvalidDataException(
                "AuraTools visual bundle material is missing pass " + requiredPass + ": " + materialPath);
        return material;
    }

    private static void ValidateCardFrameEffectUiPixels(Material source)
    {
        if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Null)
            throw new InvalidOperationException(
                "CardFrameEffectUI render validation requires a real graphics device, not NullGfx.");

        var previousActive = RenderTexture.active;
        RenderTexture target = null;
        Texture2D sourceTexture = null;
        Texture2D readback = null;
        Sprite sprite = null;
        Material material = null;
        GameObject cameraObject = null;
        GameObject canvasObject = null;
        try
        {
            target = new RenderTexture(128, 128, 24, RenderTextureFormat.ARGB32)
            {
                name = "AuraTools_CardFrameEffectUI_SmokeTarget"
            };
            if (!target.Create())
                throw new InvalidOperationException("CardFrameEffectUI smoke RenderTexture could not be created.");

            cameraObject = new GameObject("AuraTools_CardFrameEffectUI_SmokeCamera", typeof(Camera));
            var camera = cameraObject.GetComponent<Camera>();
            camera.clearFlags = CameraClearFlags.SolidColor;
            camera.backgroundColor = Color.black;
            camera.orthographic = true;
            camera.orthographicSize = 64f;
            camera.nearClipPlane = 0.01f;
            camera.farClipPlane = 100f;
            camera.transform.position = new Vector3(0f, 0f, -10f);
            camera.targetTexture = target;

            canvasObject = new GameObject(
                "AuraTools_CardFrameEffectUI_SmokeCanvas",
                typeof(RectTransform),
                typeof(Canvas),
                typeof(CanvasScaler));
            var canvas = canvasObject.GetComponent<Canvas>();
            canvas.renderMode = RenderMode.ScreenSpaceCamera;
            canvas.worldCamera = camera;
            canvas.planeDistance = 1f;

            var imageObject = new GameObject(
                "AuraTools_CardFrameEffectUI_SmokeImage",
                typeof(RectTransform),
                typeof(CanvasRenderer),
                typeof(Image));
            imageObject.transform.SetParent(canvasObject.transform, false);
            var rect = imageObject.GetComponent<RectTransform>();
            rect.anchorMin = new Vector2(0.25f, 0.25f);
            rect.anchorMax = new Vector2(0.75f, 0.75f);
            rect.offsetMin = Vector2.zero;
            rect.offsetMax = Vector2.zero;

            sourceTexture = new Texture2D(2, 2, TextureFormat.RGBA32, false, false)
            {
                name = "AuraTools_CardFrameEffectUI_SmokeSource"
            };
            sourceTexture.SetPixels(new[] { Color.white, Color.white, Color.white, Color.white });
            sourceTexture.Apply(false, false);
            sprite = Sprite.Create(
                sourceTexture,
                new Rect(0f, 0f, sourceTexture.width, sourceTexture.height),
                new Vector2(0.5f, 0.5f),
                100f);

            material = new Material(source) { name = "AuraTools_CardFrameEffectUI_SmokeMaterial" };
            material.SetFloat("_TerriasEffectIntensity", 0f);
            material.SetFloat("_TerriasOverlayMode", 0f);
            var image = imageObject.GetComponent<Image>();
            image.sprite = sprite;
            image.color = Color.white;
            image.material = material;

            Canvas.ForceUpdateCanvases();
            camera.Render();

            RenderTexture.active = target;
            readback = new Texture2D(1, 1, TextureFormat.RGBA32, false, false);
            readback.ReadPixels(new Rect(64f, 64f, 1f, 1f), 0, 0, false);
            readback.Apply(false, false);
            var pixel = readback.GetPixel(0, 0);
            var magenta = pixel.r > 0.8f && pixel.g < 0.2f && pixel.b > 0.8f;
            var empty = pixel.r < 0.1f && pixel.g < 0.1f && pixel.b < 0.1f;
            if (magenta || empty)
                throw new InvalidDataException(
                    "CardFrameEffectUI render smoke produced an invalid center pixel: " + pixel);
            Debug.Log(
                "CardFrameEffectUI render smoke passed: rgba=" + pixel
                + ", graphics=" + SystemInfo.graphicsDeviceType + ".");
        }
        finally
        {
            RenderTexture.active = previousActive;
            if (canvasObject != null) UnityEngine.Object.DestroyImmediate(canvasObject);
            if (cameraObject != null) UnityEngine.Object.DestroyImmediate(cameraObject);
            if (material != null) UnityEngine.Object.DestroyImmediate(material);
            if (sprite != null) UnityEngine.Object.DestroyImmediate(sprite);
            if (sourceTexture != null) UnityEngine.Object.DestroyImmediate(sourceTexture);
            if (readback != null) UnityEngine.Object.DestroyImmediate(readback);
            if (target != null)
            {
                target.Release();
                UnityEngine.Object.DestroyImmediate(target);
            }
        }
    }

    private static void ValidateCardFrameEffectUrpPixels(Material source)
    {
        if (SystemInfo.graphicsDeviceType == GraphicsDeviceType.Null)
            throw new InvalidOperationException(
                "CardFrameEffectURP render validation requires a real graphics device, not NullGfx.");

        var previousActive = RenderTexture.active;
        RenderTexture target = null;
        Texture2D sourceTexture = null;
        Texture2D readback = null;
        Material material = null;
        Material nativeMaterial = null;
        Mesh mesh = null;
        GameObject cameraObject = null;
        GameObject canvasObject = null;
        GameObject meshObject = null;
        try
        {
            target = new RenderTexture(128, 128, 24, RenderTextureFormat.ARGB32)
            {
                name = "AuraTools_CardFrameEffectURP_SmokeTarget"
            };
            if (!target.Create())
                throw new InvalidOperationException("CardFrameEffectURP smoke RenderTexture could not be created.");

            cameraObject = new GameObject("AuraTools_CardFrameEffectURP_SmokeCamera", typeof(Camera));
            var camera = cameraObject.GetComponent<Camera>();
            camera.clearFlags = CameraClearFlags.SolidColor;
            camera.backgroundColor = Color.black;
            camera.orthographic = true;
            camera.orthographicSize = 1f;
            camera.nearClipPlane = 0.01f;
            camera.farClipPlane = 100f;
            camera.transform.position = new Vector3(0f, 0f, -10f);
            camera.targetTexture = target;

            canvasObject = new GameObject(
                "AuraTools_CardFrameEffectURP_SmokeWorldCanvas",
                typeof(RectTransform),
                typeof(Canvas));
            canvasObject.GetComponent<Canvas>().renderMode = RenderMode.WorldSpace;

            meshObject = new GameObject(
                "AuraTools_CardFrameEffectURP_SmokeMesh",
                typeof(MeshFilter),
                typeof(MeshRenderer));
            meshObject.transform.SetParent(canvasObject.transform, false);
            mesh = new Mesh { name = "AuraTools_CardFrameEffectURP_SmokeQuad" };
            mesh.vertices = new[]
            {
                new Vector3(-0.75f, -0.75f, 0f),
                new Vector3(0.75f, -0.75f, 0f),
                new Vector3(0.75f, 0.75f, 0f),
                new Vector3(-0.75f, 0.75f, 0f)
            };
            mesh.uv = new[]
            {
                new Vector2(0f, 0f),
                new Vector2(1f, 0f),
                new Vector2(1f, 1f),
                new Vector2(0f, 1f)
            };
            mesh.triangles = new[] { 0, 2, 1, 0, 3, 2 };
            mesh.RecalculateBounds();
            meshObject.GetComponent<MeshFilter>().sharedMesh = mesh;

            sourceTexture = new Texture2D(2, 2, TextureFormat.RGBA32, false, false)
            {
                name = "AuraTools_CardFrameEffectURP_SmokeSource"
            };
            sourceTexture.SetPixels(new[] { Color.white, Color.white, Color.white, Color.white });
            sourceTexture.Apply(false, false);

            var renderer = meshObject.GetComponent<MeshRenderer>();
            material = CreateCardFrameEffectUrpSmokeMaterial(source, sourceTexture, "Initial");
            renderer.sharedMaterial = material;
            readback = new Texture2D(1, 1, TextureFormat.RGBA32, false, false);
            var pixel = ReadCenterPixel(camera, target, readback);
            ValidatePixel("CardFrameEffectURP render smoke", pixel);
            Debug.Log(
                "CardFrameEffectURP render smoke passed: rgba=" + pixel
                + ", graphics=" + SystemInfo.graphicsDeviceType + ".");

            var nativeShader = Shader.Find("Universal Render Pipeline/Unlit");
            if (nativeShader == null || !nativeShader.isSupported)
                throw new InvalidDataException("CardFrameEffectURP lease smoke cannot resolve the native URP Unlit shader.");
            nativeMaterial = new Material(nativeShader)
            {
                name = "AuraTools_CardFrameEffectURP_SmokeNativeMaterial",
                mainTexture = sourceTexture
            };
            if (nativeMaterial.HasProperty("_BaseMap")) nativeMaterial.SetTexture("_BaseMap", sourceTexture);
            renderer.sharedMaterial = nativeMaterial;
            UnityEngine.Object.DestroyImmediate(material);
            material = null;
            var restoredPixel = ReadCenterPixel(camera, target, readback);
            ValidatePixel("CardFrameEffectURP restored material smoke", restoredPixel);

            material = CreateCardFrameEffectUrpSmokeMaterial(source, sourceTexture, "Rebound");
            renderer.sharedMaterial = material;
            var reboundPixel = ReadCenterPixel(camera, target, readback);
            ValidatePixel("CardFrameEffectURP rebound material smoke", reboundPixel);
            Debug.Log(
                "CardFrameEffectURP material lease smoke passed: restored=" + restoredPixel
                + ", rebound=" + reboundPixel
                + ", graphics=" + SystemInfo.graphicsDeviceType + ".");
        }
        finally
        {
            RenderTexture.active = previousActive;
            if (meshObject != null) UnityEngine.Object.DestroyImmediate(meshObject);
            if (canvasObject != null) UnityEngine.Object.DestroyImmediate(canvasObject);
            if (cameraObject != null) UnityEngine.Object.DestroyImmediate(cameraObject);
            if (material != null) UnityEngine.Object.DestroyImmediate(material);
            if (nativeMaterial != null) UnityEngine.Object.DestroyImmediate(nativeMaterial);
            if (mesh != null) UnityEngine.Object.DestroyImmediate(mesh);
            if (sourceTexture != null) UnityEngine.Object.DestroyImmediate(sourceTexture);
            if (readback != null) UnityEngine.Object.DestroyImmediate(readback);
            if (target != null)
            {
                target.Release();
                UnityEngine.Object.DestroyImmediate(target);
            }
        }
    }

    private static Material CreateCardFrameEffectUrpSmokeMaterial(
        Material source,
        Texture sourceTexture,
        string suffix)
    {
        var material = new Material(source)
        {
            name = "AuraTools_CardFrameEffectURP_SmokeMaterial_" + suffix
        };
        material.SetTexture("_MainTex", sourceTexture);
        material.SetTexture("_NoiseTex", sourceTexture);
        material.SetTexture("_FoilTex", sourceTexture);
        material.SetFloat("_TerriasEffectIntensity", 0f);
        material.SetFloat("_TerriasOverlayMode", 0f);
        return material;
    }

    private static Color ReadCenterPixel(Camera camera, RenderTexture target, Texture2D readback)
    {
        camera.Render();
        RenderTexture.active = target;
        readback.ReadPixels(new Rect(64f, 64f, 1f, 1f), 0, 0, false);
        readback.Apply(false, false);
        return readback.GetPixel(0, 0);
    }

    private static void ValidatePixel(string scope, Color pixel)
    {
        var magenta = pixel.r > 0.8f && pixel.g < 0.2f && pixel.b > 0.8f;
        var empty = pixel.r < 0.1f && pixel.g < 0.1f && pixel.b < 0.1f;
        if (magenta || empty)
            throw new InvalidDataException(scope + " produced an invalid center pixel: " + pixel);
    }

    private static Material EnsureMaterial(string shaderPath, string materialPath, string name)
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(shaderPath);
        if (shader == null) throw new FileNotFoundException("AuraTools shader is missing.", shaderPath);
        var errors = ShaderUtil.GetShaderMessages(shader)
            .Where(message => message.severity == ShaderCompilerMessageSeverity.Error)
            .Select(message => message.message)
            .ToArray();
        if (errors.Length > 0)
            throw new InvalidOperationException("AuraTools shader compile failed: " + shaderPath + " -> " + string.Join("; ", errors));
        Directory.CreateDirectory(Path.GetDirectoryName(materialPath));
        var material = AssetDatabase.LoadAssetAtPath<Material>(materialPath);
        if (material == null)
        {
            material = new Material(shader) { name = name };
            AssetDatabase.CreateAsset(material, materialPath);
        }
        else
        {
            material.shader = shader;
        }
        return material;
    }

    private static void AssignBundleName(string materialPath)
    {
        var importer = AssetImporter.GetAtPath(materialPath);
        if (importer == null) throw new FileNotFoundException("AuraTools material importer is missing.", materialPath);
        importer.assetBundleName = BundleName;
        importer.SaveAndReimport();
    }
}
