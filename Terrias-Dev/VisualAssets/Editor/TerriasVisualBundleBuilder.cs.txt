// Copy this file into a Unity Editor project as Assets/Editor/TerriasVisualBundleBuilder.cs.
// Keep the asset paths aligned with Terrias-Dev/VisualAssets/terrias_visuals.pipeline.json
// and Terrias/visual.registry.json.

using System;
using System.IO;
using UnityEditor;
using UnityEngine;

public static class TerriasVisualBundleBuilder
{
    private const string BundleName = "terrias_visuals";
    private const string StarScoreShaderAssetPath = "Assets/Terrias/Visuals/Shaders/StarScoreHud.shader";
    private const string MaterialAssetPath = "Assets/Terrias/Visuals/Materials/StarScoreHudLit.mat";
    private const string StarScoreMaterialAssetPath = MaterialAssetPath;
    private const string WunaOrbitFireShaderAssetPath = "Assets/Terrias/Visuals/Shaders/WunaOrbitFire.shader";
    private const string WunaOrbitFireBackMaterialAssetPath = "Assets/Terrias/Visuals/Materials/WunaOrbitFireBack.mat";
    private const string WunaOrbitFireFrontMaterialAssetPath = "Assets/Terrias/Visuals/Materials/WunaOrbitFireFront.mat";
    private const string CardUseStardustShaderAssetPath = "Assets/Terrias/Visuals/Shaders/CardUseStardust.shader";
    private const string CardUseStardustMaterialAssetPath = "Assets/Terrias/Visuals/Materials/CardUseStardust.mat";
    private const string BuildOutputDirectory = "Build/TerriasVisualBundles";
    private const string ModOutputPath = "../Terrias/ModResource/VisualBundles/terrias_visuals";

    [MenuItem("Terrias/Visuals/Build Visual Bundle")]
    public static void BuildVisualBundle()
    {
        EnsureStarScoreMaterial();
        EnsureWunaOrbitFireMaterials();
        EnsureCardUseStardustMaterial();

        var starScoreMaterial = AssetImporter.GetAtPath(StarScoreMaterialAssetPath);
        var wunaBackMaterial = AssetImporter.GetAtPath(WunaOrbitFireBackMaterialAssetPath);
        var wunaFrontMaterial = AssetImporter.GetAtPath(WunaOrbitFireFrontMaterialAssetPath);
        var cardUseStardustMaterial = AssetImporter.GetAtPath(CardUseStardustMaterialAssetPath);
        if (starScoreMaterial == null || wunaBackMaterial == null || wunaFrontMaterial == null || cardUseStardustMaterial == null)
        {
            throw new FileNotFoundException("One or more Terrias material assets are missing.");
        }

        AssignBundleName(starScoreMaterial);
        AssignBundleName(wunaBackMaterial);
        AssignBundleName(wunaFrontMaterial);
        AssignBundleName(cardUseStardustMaterial);
        AssetDatabase.RemoveUnusedAssetBundleNames();
        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh(ImportAssetOptions.ForceSynchronousImport);

        var bundleAssets = new[]
        {
            StarScoreMaterialAssetPath,
            WunaOrbitFireBackMaterialAssetPath,
            WunaOrbitFireFrontMaterialAssetPath,
            CardUseStardustMaterialAssetPath
        };
        var builds = new[]
        {
            new AssetBundleBuild
            {
                assetBundleName = BundleName,
                assetNames = bundleAssets
            }
        };

        Directory.CreateDirectory(BuildOutputDirectory);
        BuildPipeline.BuildAssetBundles(
            BuildOutputDirectory,
            builds,
            BuildAssetBundleOptions.ChunkBasedCompression,
            EditorUserBuildSettings.activeBuildTarget);

        var builtBundle = Path.Combine(BuildOutputDirectory, BundleName);
        if (!File.Exists(builtBundle))
        {
            throw new FileNotFoundException("Terrias visual bundle was not produced.", builtBundle);
        }

        var outputPath = ResolveModOutputPath();
        Directory.CreateDirectory(Path.GetDirectoryName(outputPath));
        File.Copy(builtBundle, outputPath, true);
        AssetDatabase.Refresh();
        Debug.Log("Built Terrias visual bundle: " + outputPath);
    }

    private static string ResolveModOutputPath()
    {
        var explicitPath = Environment.GetEnvironmentVariable("TERRIAS_VISUAL_BUNDLE_OUTPUT");
        return string.IsNullOrWhiteSpace(explicitPath) ? ModOutputPath : explicitPath;
    }

    private static void AssignBundleName(AssetImporter importer)
    {
        if (importer.assetBundleName == BundleName)
        {
            return;
        }

        importer.assetBundleName = BundleName;
        importer.SaveAndReimport();
    }

    private static void EnsureStarScoreMaterial()
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(StarScoreShaderAssetPath);
        if (shader == null)
        {
            throw new FileNotFoundException("Terrias shader asset is missing.", StarScoreShaderAssetPath);
        }

        var material = AssetDatabase.LoadAssetAtPath<Material>(StarScoreMaterialAssetPath);
        if (material == null)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(StarScoreMaterialAssetPath));
            material = new Material(shader)
            {
                name = "StarScoreHudLit"
            };
            AssetDatabase.CreateAsset(material, StarScoreMaterialAssetPath);
        }
        else
        {
            material.shader = shader;
        }

        material.SetFloat("_TerriasFlowSpeed", 0.55f);
        material.SetFloat("_TerriasFlowScale", 1.2f);
        material.SetFloat("_TerriasEdgeGlow", 0.35f);
        material.SetColor("_TerriasGlowColor", new Color(1f, 0.878f, 0.541f, 1f));
        material.SetColor("_TerriasFlowColor", new Color(0.616f, 0.863f, 1f, 1f));
        EditorUtility.SetDirty(material);
        AssetDatabase.SaveAssets();
    }

    private static void EnsureWunaOrbitFireMaterials()
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(WunaOrbitFireShaderAssetPath);
        if (shader == null)
        {
            throw new FileNotFoundException("Terrias WuNa orbit fire shader asset is missing.", WunaOrbitFireShaderAssetPath);
        }

        var back = EnsureMaterial(WunaOrbitFireBackMaterialAssetPath, shader, "WunaOrbitFireBack");
        back.SetFloat("_TerriasLayer", -1f);
        back.SetFloat("_TerriasNoiseScale", 3.15f);
        back.SetFloat("_TerriasDistortion", 0.11f);
        back.SetFloat("_TerriasAlphaCutoff", 0.16f);
        back.SetFloat("_TerriasAlphaSoftness", 0.11f);
        back.SetFloat("_TerriasFlowSpeed", 0.28f);
        back.SetColor("_TerriasCoreColor", new Color(1f, 0.82f, 0.34f, 0.72f));
        back.SetColor("_TerriasEdgeColor", new Color(0.96f, 0.18f, 0.045f, 0.62f));
        back.SetColor("_TerriasSmokeColor", new Color(0.2f, 0.02f, 0.012f, 0.18f));

        var front = EnsureMaterial(WunaOrbitFireFrontMaterialAssetPath, shader, "WunaOrbitFireFront");
        front.SetFloat("_TerriasLayer", 1f);
        front.SetFloat("_TerriasNoiseScale", 3.8f);
        front.SetFloat("_TerriasDistortion", 0.16f);
        front.SetFloat("_TerriasAlphaCutoff", 0.15f);
        front.SetFloat("_TerriasAlphaSoftness", 0.095f);
        front.SetFloat("_TerriasFlowSpeed", 0.38f);
        front.SetColor("_TerriasCoreColor", new Color(1f, 0.96f, 0.58f, 0.96f));
        front.SetColor("_TerriasEdgeColor", new Color(1f, 0.22f, 0.035f, 0.84f));
        front.SetColor("_TerriasSmokeColor", new Color(0.24f, 0.026f, 0.014f, 0.2f));

        EditorUtility.SetDirty(back);
        EditorUtility.SetDirty(front);
        AssetDatabase.SaveAssets();
    }

    private static void EnsureCardUseStardustMaterial()
    {
        var shader = AssetDatabase.LoadAssetAtPath<Shader>(CardUseStardustShaderAssetPath);
        if (shader == null)
        {
            throw new FileNotFoundException("Terrias Star Score card-use shader asset is missing.", CardUseStardustShaderAssetPath);
        }

        var material = EnsureMaterial(CardUseStardustMaterialAssetPath, shader, "CardUseStardust");
        material.SetFloat("_TerriasEffectMode", 1f);
        material.SetFloat("_TerriasOverlayMode", 0f);
        material.SetFloat("_TerriasFrameOnlyOverlay", 0f);
        material.SetFloat("_TerriasFlowSpeed", 0.72f);
        material.SetFloat("_TerriasFlowScale", 1.5f);
        material.SetFloat("_TerriasNoiseScale", 6.2f);
        material.SetFloat("_TerriasDistortion", 0.004f);
        material.SetFloat("_TerriasEffectIntensity", 0.95f);
        material.SetFloat("_TerriasQualityScale", 1f);
        material.SetFloat("_TerriasEdgeGlow", 0.12f);
        material.SetFloat("_TerriasStardustGrain", 0.28f);
        material.SetFloat("_TerriasStardustDensity", 0.46f);
        material.SetFloat("_TerriasStardustTwinkle", 1.25f);
        material.SetFloat("_TerriasStardustTwinkleSpeed", 2.15f);
        material.SetFloat("_TerriasStardustOrbit", 0.18f);
        material.SetFloat("_TerriasStardustGlowRadius", 0.18f);
        material.SetFloat("_TerriasStardustGlowPower", 5.4f);
        material.SetFloat("_TerriasStardustSweepSpeed", 1.85f);
        material.SetFloat("_TerriasStardustSweepIntensity", 0.62f);
        material.SetFloat("_TerriasStardustSweepWidth", 0.045f);
        material.SetFloat("_TerriasEdgeSample", 2f);
        material.SetColor("_TerriasStardustColorA", new Color(0.953f, 0.984f, 1f, 1f));
        material.SetColor("_TerriasStardustColorB", new Color(1f, 0.902f, 0.659f, 1f));
        EditorUtility.SetDirty(material);
        AssetDatabase.SaveAssets();
    }

    private static Material EnsureMaterial(string path, Shader shader, string materialName)
    {
        var material = AssetDatabase.LoadAssetAtPath<Material>(path);
        if (material == null)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            material = new Material(shader)
            {
                name = materialName
            };
            AssetDatabase.CreateAsset(material, path);
        }
        else
        {
            material.shader = shader;
        }

        return material;
    }
}
